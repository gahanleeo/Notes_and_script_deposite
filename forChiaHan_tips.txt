vcf.file.tmp <- read.vcfR("../TERT-CLPTM1L_project/data_WGS30x_onGRCh38/imputing/panel_1KGP_SNV_INDEL_SV_HapsTRs_phased_panel_hg38.sort.vcf.gz")

# subset VCF for T2/haplotypes
vcf.file.set <- vcf.file.tmp[ which(getPOS(vcf.file.tmp) %in% gsub("chr5:|:[A-Z]", "", snp.list$SNP.pick_hg38)), ]

# get and format genotypes
vcf_genotypes.tmp <- extract.gt(
  vcf.file.set,
  element = "GT",
  mask = FALSE,
  as.numeric = FALSE,
  return.alleles = TRUE,
  IDtoRowNames = TRUE,
  extract = TRUE,
  convertNA = TRUE
)

geno_df.tmp <- as.data.frame(t(vcf_genotypes.tmp))

names(geno_df.tmp) <- snp.list$rsid[ match(gsub("^5:|:[A-Z]", "", names(geno_df.tmp)), gsub("chr5:|:[A-Z]", "", snp.list$SNP.pick_hg38)) ]
geno_df.tmp$ID_person <- row.names(geno_df.tmp)
geno_df.tmp <- data.frame(lapply(geno_df.tmp, function(x){ gsub("\\|", " ", x)}))



# get and format haplotypes
vcf_haplotype <- extract.haps(
  vcf.file.set, mask = FALSE, unphased_as_NA = TRUE, verbose = TRUE)
head(as.data.frame(vcf_haplotype[,1:5]))

haps_df <- as.data.frame(t(vcf_haplotype))
haps_df$haplotypes <- unite(haps_df, col = "set", sep = "", remove = TRUE, na.rm = FALSE)
haps_df$sample <- row.names(haps_df)
haps_df$ID <- str_sub(haps_df$sample, -1)
haps_df$sample <- gsub("_1$", "", haps_df$sample)
haps_df$sample <- gsub("_0$", "", haps_df$sample)
haps_df$haplotypes <- haps_df$haplotypes$set
haps_df <- haps_df %>% dplyr::select(c("sample", "ID", "haplotypes", everything()))
